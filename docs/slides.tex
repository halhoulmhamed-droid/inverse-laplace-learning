\begin{frame}{Une question comparative, un résultat négatif}
\begin{block}{Question}
Le plafonnement d'un filtre appris améliore-t-il les erreurs extrêmes sous observations partielles ou erreur de hauteur ?
\end{block}
\begin{itemize}
\item Onze méthodes affines ; aucun réseau neuronal nécessaire.
\item Données synthétiques ; générateur analytique distinct de l'inverse numérique.
\item 2\,376\,000 reconstructions, trois entraînements, zéro échec numérique détecté.
\item Le critère préfixé de gain robuste n'est satisfait dans aucun entraînement.
\end{itemize}
\vspace{2mm}
Une exécution réussie n'implique pas une supériorité statistique.
\end{frame}

\begin{frame}{Le transfert amortit exponentiellement les modes}
\begin{columns}[T]
\column{.51\textwidth}
$\Omega=(0,1)\times(0,H)$, $H=1/4$.
\begin{itemize}
\item $\Delta u=0$ ; trace supérieure $q$ inconnue.
\item Flux nul sur bas et côtés : hypothèse contrôlée.
\item 64 moyennes inférieures, éventuellement masquées.
\end{itemize}
\column{.47\textwidth}
$\phi_0=1$, $\phi_k=\sqrt2\cos(k\pi x)$.
\[
u_k(x,y)=\phi_k(x)\frac{\cosh(k\pi y)}{\cosh(k\pi H)}.
\]
\[
T_H\phi_k=\operatorname{sech}(k\pi H)\phi_k.
\]
$\norm{\phi_k}=1$ mais $\norm{T_H\phi_k}\to0$ : inverse non borné.
\end{columns}
\vfill
Le mode constant reste estimé ; le centrage statistique ne le supprime pas.
\end{frame}

\begin{frame}{La matrice inverse n'est pas le transfert continu}
\begin{columns}
\column{.43\textwidth}
\small
Capteurs : intervalles de longueur $1/64$. Masques : complet, pairs, contigu.
\[
W=\tfrac1{64}I_m,\quad A=QA_{\rm raw},
\]
\[
Q=W^{1/2}/\norm{W^{1/2}A_{\rm raw}}.
\]
Bruit après moyenne, avant masque ; calibration entraînement seulement.

Rang numérique : $\sigma_j>10^{-12}\sigma_1$.

\end{columns}
\end{frame}

\begin{frame}{Le plafond borne une amplification, pas le risque total}
\[
d=(A+E)c+b+\varepsilon,\qquad \widehat c=\mu_c+R(d-\mu_d).
\]
\begin{block}{Identité exacte, avec $f=A\mu_c-\mu_d$}
\[
\widehat c-c=(RA-I)(c-\mu_c)+R(Ec+b+\varepsilon+f).
\]
\end{block}
\begin{itemize}
\item Premier terme : biais, noyau et directions rejetées inclus.
\item Second : erreur du modèle, queue transmise, bruit et offset.
\item Erreur totale : $\norm{\widehat q-q}^2=\norm{\widehat c-c}^2+\norm{q_\perp}^2$.
\end{itemize}
\small Pour un filtre diagonal, $\norm R=\max|g_j|$ ; H1 et LMMSE demandent la vraie norme matricielle. Réduire la variance peut augmenter le biais.
\end{frame}

\begin{frame}{Ce qui est appris et ce qui est comparé}
\[
g_* =\frac{\operatorname{Cov}(\beta,z)}{\operatorname{Var}z},\qquad
z=\sigma\beta+\varepsilon\ \Longrightarrow\ g_W=\frac{\sigma p}{\sigma^2p+v}
\]
\small La dernière égalité suppose signal et bruit décorrélés. Moyennes non nulles : estimateur affine. Sous erreur du modèle, gains négatifs possibles.
\begin{columns}[T]
\column{.49\textwidth}
\begin{block}{Références}
Tikhonov L2 et H1 ; TSVD ; Wiener ; Wiener plafonné ; LMMSE complet.
\end{block}
\column{.49\textwidth}
\begin{block}{Cinq filtres appris}
Signé ; signé plafonné ; positif ; positif borné par $1/\sigma$ ; borné plafonné.
\end{block}
\end{columns}
\small H1 ne commute généralement pas avec $A^TA$. LMMSE exploite les covariances croisées et peut prédire dans le noyau, sans créer d'identifiabilité déterministe.
\end{frame}

\begin{frame}{Validation sans vérité et incertitude par objet}
\begin{itemize}
\item Trois entraînements de 1000 objets ; 200 validations ; 200 tests nominaux et 200 rugueux.
\item Dix répétitions bruitées par objet test ; objets pilote distincts.
\item Sept grilles de 20 configurations maximum, quatre méthodes fixes.
\end{itemize}
\[
Q=\frac1{N_v}\sum_i\norm{A\widehat c(d_i^{(1)})-d_i^{(2)}}^2.
\]
\small Deux répétitions indépendantes conditionnellement à l'objet. Le risque prédictif n'est pas le risque inverse : le noyau reste peu visible.
\begin{block}{Critère préfixé}
À 1\,\% de bruit : P1 contigu nominal ; P2 complet, hauteur +5\,\%. Borne du ratio q95 $<0,90$ et borne du ratio de médiane nominale $<1,05$.
\end{block}
\tiny Bootstrap : 2000 tirages d'objets, quatre bornes à 98,75\,\% par entraînement. Pas de couverture simultanée des trois entraînements.
\end{frame}
