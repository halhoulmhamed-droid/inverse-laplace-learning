\begin{frame}{Question et réponse limitée}
\begin{block}{Question préspécifiée}
Les covariances complètes améliorent-elles les covariances diagonales dans les mêmes 12 directions singulières, avec mêmes données, moyennes et réglage ?
\end{block}
\par\vspace{3mm}
\begin{itemize}
\item Oui selon la règle figée : risque F/D proche de 0,32 sur 32 capteurs contigus, $N=256$.
\item LMMSE complet bénéficie aussi de la prédiction hors du sous-espace : effet distinct.
\item Dégradations observées face à H1 avec tous les capteurs et sur des objets plus rugueux.
\end{itemize}
\par\vspace{2mm}
\small Les résultats de l'étude 1 ont formulé l'hypothèse ; ses objets de test ne sont pas réutilisés. Sa conclusion négative reste intégralement conservée.
\end{frame}

\begin{frame}{Modèle : indépendance dans la base cosinus}
\[
q(x)=\sum_{k=0}^{127}c_k\phi_k(x),\quad c_k=w_k\xi_k,\quad
\xi_k\ \text{indépendants},\quad w_k^2\propto(1+k^2)^{-p}.
\]
\begin{itemize}
\item Rectangle harmonique, $H=0,25$, 64 moyennes de capteurs ; transfert $\operatorname{sech}(k\pi H)$.
\item Générateur analytique 128 modes ; inverse FD à 64 modes, grille $64\times16$ intervalles.
\item Bruit absolu gaussien ; observations divisées par 8, poids $1/64$ conservés après masque.
\end{itemize}
\begin{block}{Attention à l'interprétation}
$\operatorname{Cov}(V_r^\top c)=V_r^\top D_cV_r$ peut être non diagonale. Changement de base et capteurs partiels ne prouvent pas des corrélations physiques des coefficients cosinus.
\end{block}
\end{frame}

\begin{frame}{Cinq restrictions pour séparer les effets}
\small $z=U_r^\top(y-\mu_y)$, $b=V_r^\top(c-\mu_c)$, $r=12$, $P=V_rV_r^\top$. Toujours $\widehat c=\mu_c+R(y-\mu_y)$.
\par\vspace{3mm}
\begin{center}
\begin{tabular}{llll}
\toprule & Entrée & Sortie de l'incrément & Covariances\\\midrule
D & $U_r$ & $V_r$ & Diagonales $C_{bz},C_{zz}$\\
F & $U_r$ & $V_r$ & Blocs complets\\
O & $U_r$ & Toute la sortie & Blocs complets\\
L & Toutes les observations & Toute la sortie & Blocs complets\\
LP & Toutes les observations & $V_r$ & Projection de L\\\bottomrule
\end{tabular}
\end{center}
\par\vspace{3mm}
\begin{block}{Identités testées à même ridge}
$PR_O=R_F$ et $PR_L=R_{LP}$. Projeter l'incrément, pas la moyenne. Les diagnostics utilisent $\lambda_F$ pour D/F/O et $\lambda_L$ pour L/LP.
\end{block}
\end{frame}

\begin{frame}{Pourquoi l'ablation est mathématiquement contrôlée}
\[
\min_G\;\mathbb E\|b-Gz\|^2+\lambda t\|G\|_F^2
\quad\Longrightarrow\quad G=C_{bz}(C_{zz}+\lambda tI)^{-1}.
\]
\begin{itemize}
\item Contrainte $G$ diagonale : $g_j=C_{bz,jj}/(C_{zz,jj}+\lambda t)$.
\item Même base, même sous-espace, même échelle $t$, mêmes moyennes, 20 candidats chacun.
\item Version empirique régularisée de LMMSE ; aucune optimalité universelle sur le test.
\end{itemize}
\[
\|\widehat q-q\|^2=\|Pe\|^2+\|(I-P)e\|^2+\sum_{k=64}^{127}c_k^2.
\]
\small À ridge commun, F et O ont exactement la même perte dans $P$. Leur écart total provient du complément, qui n'est pas nécessairement le noyau exact de l'opérateur.
\end{frame}
